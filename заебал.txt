function FindProxyForURL(url, host) {
    // 1. Прямой доступ для локальной сети и localhost
    if (isPlainHostName(host) || 
        shExpMatch(host, "localhost") || 
        shExpMatch(host, "127.*") || 
        shExpMatch(host, "192.168.*") || 
        shExpMatch(host, "10.*")) {
        return "DIRECT";
    }

    // 2. Прямой доступ для российских доменов и VK
    if (shExpMatch(host, "*.ru") || 
        shExpMatch(host, "*.su") || 
        shExpMatch(host, "*.рф") || 
        dnsDomainIs(host, "vk.com") || 
        dnsDomainIs(host, "vkontakte.ru")) {
        return "DIRECT";
    }

    // 3. Всё остальное — через прокси
    return "PROXY 148.253.210.145:443";
}
